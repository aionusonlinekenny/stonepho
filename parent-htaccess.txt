# Enable Rewrite Engine
RewriteEngine On

# Redirect www to non-www (stonephovaldosta.com has SSL, www.stonephovaldosta.com doesn't)
RewriteCond %{HTTP_HOST} ^www\.stonephovaldosta\.com [NC]
RewriteRule ^(.*)$ https://stonephovaldosta.com/$1 [L,R=301]

# Redirect all HTTP to HTTPS
RewriteCond %{HTTPS} !=on
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Disable caching for versions.php
<Files "versions.php">
  Header set Cache-Control "no-store, no-cache, must-revalidate, max-age=0"
  Header set Pragma "no-cache"
  Header set Expires "0"
</Files>

# Headers - merge into single block
<IfModule mod_headers.c>
  # Remove old CSP headers
  Header always unset Content-Security-Policy
  Header always unset X-Content-Security-Policy
  Header always unset X-WebKit-CSP

  # Relaxed CSP to allow uploads and API calls
  Header always set Content-Security-Policy "default-src 'self'; connect-src 'self' https://stonephovaldosta.com; img-src 'self' data: https:; style-src 'self' 'unsafe-inline'; script-src 'self' 'unsafe-inline' 'unsafe-eval'; frame-src 'self' https://www.google.com https://maps.google.com"

  # CORS for APIs
  Header always set Access-Control-Allow-Origin "*"
  Header always set Access-Control-Allow-Methods "GET, POST, PUT, DELETE, OPTIONS"
  Header always set Access-Control-Allow-Headers "Content-Type, Authorization, X-Requested-With"

  # Security headers
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-XSS-Protection "1; mode=block"
</IfModule>
